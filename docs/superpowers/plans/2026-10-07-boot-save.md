# Boot Save Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `FABLE2_BOOT_SAVE=Hero003` boots straight into that save with no menu input, and `FABLE2_SKIP_LOGOS=1` drops the boot logo videos, cutting about 30 s from each test launch.

**Architecture:** The front-end menu is only UI over one global, the initial save name at guest `0x834973BC`. The Load Game handler (`sub_82695590`) validates the save, writes the name and calls `sub_826962D0(frontEnd, userIndex)`. We override the per-frame front-end update (`sub_82692230`, `r3` = front-end object) with a strong C++ definition, and once the front end accepts input we run the same sequence. No manifest or codegen change: recompiled functions are weak aliases of `__imp__<name>`, so a strong `extern "C"` definition replaces them (pattern: `src/core/fable2_f5_lua.h`).

**Tech Stack:** C++23 (clang), ReXGlue recompiled PPC code, PowerShell 7 driver, Python `unittest`.

**Spec:** none; the design is the "Design facts" section below (from static analysis of the game image on 2026-10-07).

```cpp
// fable_2 - Boot save: pure logic (std only). The runtime side lives in
// src/core/fable2_boot_save.h; the driver side in tools/drive_game.ps1.
//
// FABLE2_BOOT_SAVE=Hero003 loads that save from the front end without menu input.
// FABLE2_SKIP_LOGOS=1 drops the boot logo videos. FABLE2_BOOT_TRACE=1 logs front-end
// state changes. All unset: nothing changes.
//
// Guest addresses are valid for code group "goty-identical-payload-2026-09-28"
// (constants/game_versions.h; fable2_boot_save.h asserts it at compile time).

#pragma once

#include <cctype>
#include <cstdint>
#include <filesystem>
#include <optional>
#include <string>
#include <string_view>
#include <system_error>

namespace fable2::bootsave {

inline constexpr uint32_t kStartGameFn = 0x826962D0;      // (front end, user index)
inline constexpr uint32_t kSaveQueryCtorFn = 0x824E4E30;  // (query); strings at +12 and +16
inline constexpr uint32_t kSaveExistsFn = 0x824EA7A0;     // (save manager, query, name, user) -> bool
inline constexpr uint32_t kStringCtorFn = 0x8222CF18;     // (dest, cstr, len = -1)
inline constexpr uint32_t kStringAssignFn = 0x82265160;   // (dest, src)
inline constexpr uint32_t kStringDtorFn = 0x82214F08;     // (str)
inline constexpr uint32_t kInitialSaveName = 0x834973BC;  // game string
inline constexpr uint32_t kLoadClearedByte = 0x8349693F;  // the Load Game handler stores 0 here
inline constexpr uint32_t kFrontEndState = 0x83496930;    // u32
inline constexpr uint32_t kGameRoot = 0x83496928;         // save manager = *(*(*root + 12) + 40)

// The front-end input dispatcher (sub_82692048) handles input only in this state range.
inline constexpr uint32_t kFireStateMin = 3;
inline constexpr uint32_t kFireStateMax = 8;
inline constexpr uint32_t kStableUpdates = 2;  // consecutive front-end updates in range before firing
inline constexpr uint32_t kUserIndex = 0;

// Set once at startup, before any guest thread runs; read-only afterwards.
struct Config {
  std::string hero;  // empty: not armed
  bool skip_logos = false;
  bool trace = false;
  std::filesystem::path saves_root;
};
inline Config& config() {
  static Config c;
  return c;
}

// "Hero" + exactly three digits (surrounding spaces ignored, case canonicalised to "HeroNNN").
inline std::optional<std::string> ParseHeroName(std::string_view s) {
  while (!s.empty() && std::isspace(static_cast<unsigned char>(s.front()))) s.remove_prefix(1);
  while (!s.empty() && std::isspace(static_cast<unsigned char>(s.back()))) s.remove_suffix(1);
  if (s.size() != 7) return std::nullopt;
  static constexpr std::string_view kPrefix = "hero";
  for (size_t i = 0; i < 4; ++i)
    if (std::tolower(static_cast<unsigned char>(s[i])) != kPrefix[i]) return std::nullopt;
  for (size_t i = 4; i < 7; ++i)
    if (s[i] < '0' || s[i] > '9') return std::nullopt;
  return "Hero" + std::string(s.substr(4));
}

enum class Action { kWait, kFire, kDone };

// Decides when to start the load: once, after kStableUpdates consecutive front-end
// updates with the state in range. Touched only by the guest thread that runs the update.
struct Trigger {
  uint32_t stable = 0;
  bool fired = false;
  uint64_t updates = 0;

  Action Step(uint32_t fe_state) {
    ++updates;
    if (fired) return Action::kDone;
    if (fe_state < kFireStateMin || fe_state > kFireStateMax) {
      stable = 0;
      return Action::kWait;
    }
    if (++stable < kStableUpdates) return Action::kWait;
    fired = true;
    return Action::kFire;
  }
};

struct FindResult {
  std::filesystem::path dir;  // empty on failure
  std::string error;          // empty on success
};

// <saves_root>/<profile>/4D5307F1/00000001/<hero>/mainsave.bin, in exactly one profile.
inline FindResult FindHeroDir(const std::filesystem::path& saves_root, std::string_view hero) {
  FindResult out;
  std::error_code ec;
  std::filesystem::directory_iterator it(saves_root, ec);
  if (ec) {
    out.error = "saves folder not readable: " + saves_root.string();
    return out;
  }
  int found = 0;
  std::filesystem::path first;
  for (const auto& profile : it) {
    const std::filesystem::path dir = profile.path() / "4D5307F1" / "00000001" / hero;
    if (std::filesystem::is_regular_file(dir / "mainsave.bin", ec)) {
      if (found++ == 0) first = dir;
    }
  }
  if (found == 0) {
    out.error = "no save folder " + std::string(hero) + " under " + saves_root.string();
  } else if (found > 1) {
    out.error = "save folder " + std::string(hero) + " exists under " + std::to_string(found) +
                " profiles; keep one";
  } else {
    out.dir = first;
  }
  return out;
}

}  // namespace fable2::bootsave
```

```cpp
// fable_2 - Boot save: runtime side (see src/core/boot_save.h for the options).
//
// The front-end menu is UI over one global, the initial save name. Its Load Game
// handler (sub_82695590) checks that the save exists, assigns the name, clears a
// byte and calls sub_826962D0(front end, user index). OnFrontEndUpdate runs the
// same sequence from the per-frame front-end update (sub_82692230, r3 = front
// end) once the front end accepts input, so no pad input is needed.
//
// Recompiled guest functions are weak aliases of __imp__<name>; the strong
// definitions below replace them for direct and indirect calls.

#pragma once

#include <cstdint>
#include <cstring>
#include <string>

#include <rex/logging/macros.h>
#include "rex/ppc/context.h"

#include "../../constants/game_versions.h"
#include "boot_save.h"

static_assert(fable2::versions::kCodeGroup == "goty-identical-payload-2026-09-28",
              "boot save guest addresses were taken from this code group; re-derive them");

extern "C" void __imp__sub_82692230(PPCContext& ctx, uint8_t* base);
extern "C" void __imp__sub_822F4958(PPCContext& ctx, uint8_t* base);

namespace fable2::bootsave {

namespace ta = fable2::textappend;

inline uint32_t GuestZeroed(PPCContext& ctx, uint8_t* base, size_t size) {
  const uint32_t p = ta::guest_alloc(ctx, base, size);
  if (!p || !ta::wr2(base, p, size)) return 0;
  std::memset(reinterpret_cast<void*>(ta::host_addr(base, p)), 0, size);
  return p;
}

// *(*(*kGameRoot + 12) + 40), or 0 when any link is not readable.
inline uint32_t SaveManager(const uint8_t* base) {
  if (!ta::rdable2(base, kGameRoot, 4)) return 0;
  const uint32_t root = ta::load_be(base, kGameRoot);
  if (!root || !ta::rdable2(base, root + 12, 4)) return 0;
  const uint32_t mid = ta::load_be(base, root + 12);
  if (!mid || !ta::rdable2(base, mid + 40, 4)) return 0;
  return ta::load_be(base, mid + 40);
}

// The Load Game handler's sequence. Returns false (and starts nothing) when the game
// says the save does not exist. Must run on the guest thread that updates the front end.
inline bool StartLoad(PPCContext& ctx, uint8_t* base, uint32_t front_end, const std::string& hero) {
  const uint32_t cstr = GuestZeroed(ctx, base, hero.size() + 1);
  const uint32_t name = GuestZeroed(ctx, base, 16);
  const uint32_t query = GuestZeroed(ctx, base, 96);
  const uint32_t manager = SaveManager(base);
  if (!cstr || !name || !query || !manager) {
    REXSYS_WARN("[boot-save] not started: cstr={:08X} name={:08X} query={:08X} manager={:08X}",
                cstr, name, query, manager);
    return false;
  }
  std::memcpy(reinterpret_cast<void*>(ta::host_addr(base, cstr)), hero.data(), hero.size());

  ctx.r3.u32 = name;
  ctx.r4.u32 = cstr;
  ctx.r5.u32 = 0xFFFFFFFF;
  REX_CALL_INDIRECT_FUNC(kStringCtorFn);

  ctx.r3.u32 = query;
  REX_CALL_INDIRECT_FUNC(kSaveQueryCtorFn);
  ctx.r3.u32 = manager;
  ctx.r4.u32 = query;
  ctx.r5.u32 = name;
  ctx.r6.u32 = kUserIndex;
  REX_CALL_INDIRECT_FUNC(kSaveExistsFn);
  const bool exists = (ctx.r3.u32 & 0xFF) != 0;
  ctx.r3.u32 = query + 16;
  REX_CALL_INDIRECT_FUNC(kStringDtorFn);
  ctx.r3.u32 = query + 12;
  REX_CALL_INDIRECT_FUNC(kStringDtorFn);

  if (exists) {
    ctx.r3.u32 = kInitialSaveName;
    ctx.r4.u32 = name;
    REX_CALL_INDIRECT_FUNC(kStringAssignFn);
    *reinterpret_cast<uint8_t*>(ta::host_addr(base, kLoadClearedByte)) = 0;
    ctx.r3.u32 = front_end;
    ctx.r4.u32 = kUserIndex;
    REX_CALL_INDIRECT_FUNC(kStartGameFn);
  }
  ctx.r3.u32 = name;
  REX_CALL_INDIRECT_FUNC(kStringDtorFn);
  // The C string and the two small blocks are left allocated (once per run).
  return exists;
}

// Called after every front-end update with its object.
inline void OnFrontEndUpdate(PPCContext& ctx, uint8_t* base, uint32_t front_end) {
  const Config& cfg = config();
  if (!cfg.trace && cfg.hero.empty()) return;
  if (!front_end || !ta::rdable2(base, front_end + 4, 4) || !ta::rdable2(base, kFrontEndState, 4)) return;
  const uint32_t state = ta::load_be(base, kFrontEndState);
  const uint32_t screen = ta::load_be(base, front_end + 4);

  static Trigger trigger;
  static uint32_t last_state = 0xFFFFFFFF, last_screen = 0xFFFFFFFF;
  if (state != last_state || screen != last_screen) {
    REXSYS_INFO("[boot-save] front end state {} screen {} (update {})", state, screen, trigger.updates);
    last_state = state;
    last_screen = screen;
  }
  if (cfg.hero.empty()) {
    ++trigger.updates;
    return;
  }

  const Action action = trigger.Step(state);
  if (action == Action::kWait && trigger.updates == 1800)
    REXSYS_WARN("[boot-save] still waiting after 1800 front-end updates (state {}, needs {}..{})",
                state, kFireStateMin, kFireStateMax);
  if (action != Action::kFire) return;

  const FindResult save = FindHeroDir(cfg.saves_root, cfg.hero);
  if (save.dir.empty()) {
    REXSYS_WARN("[boot-save] not started: {}", save.error);
    return;
  }
  // StartLoad calls guest functions that clobber the registers; the update we hooked
  // has already returned, so restore the context it left.
  const PPCContext saved = ctx;
  const bool started = StartLoad(ctx, base, front_end, cfg.hero);
  ctx = saved;
  if (started)
    REXSYS_INFO("[boot-save] loading {} (front end state {}, update {})", cfg.hero, state, trigger.updates);
  else
    REXSYS_WARN("[boot-save] the game reports no save named {}; menu left as is", cfg.hero);
}

}  // namespace fable2::bootsave
```
